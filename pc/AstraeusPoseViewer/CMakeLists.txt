cmake_minimum_required(VERSION 3.20)
project(AstraeusPoseViewer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(fusion_protocol_tests tests/fusion_protocol_tests.cpp)
target_include_directories(fusion_protocol_tests PRIVATE include)
add_test(NAME fusion_protocol COMMAND fusion_protocol_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../protocol")
add_executable(protocol_tests tests/protocol_tests.cpp)
target_include_directories(protocol_tests PRIVATE include)
add_test(NAME protocol COMMAND protocol_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../protocol/golden_pose.hex")
if(WIN32)
    add_executable(receiver_tests tests/receiver_tests.cpp src/receiver.cpp)
    target_include_directories(receiver_tests PRIVATE include)
    target_compile_definitions(receiver_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(receiver_tests PRIVATE ws2_32)
    add_test(NAME receiver COMMAND receiver_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../protocol/golden_pose.hex")
    add_executable(AstraeusPoseViewer WIN32 src/main.cpp src/receiver.cpp src/visualization.cpp)
    target_include_directories(AstraeusPoseViewer PRIVATE include)
    target_compile_definitions(AstraeusPoseViewer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(AstraeusPoseViewer PRIVATE ws2_32 gdi32 user32)
endif()
